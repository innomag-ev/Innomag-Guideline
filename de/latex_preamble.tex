\newcommand{\innomag@addcontentsline}[3]{%
  \ifstrequal{#1}{toc}{%
    \ifstrequal{#2}{chapter}{%
      \IfBeginWith{\@currentHref}{chapter.}{}{%
        \phantomsection
      }%
    }{}%
  }{}%
  \innomag@originaladdcontentsline{#1}{#2}{#3}%
}

\AtBeginDocument{%
  \let\innomag@originaladdcontentsline\addcontentsline
  \let\addcontentsline\innomag@addcontentsline
}

% Background colors
%\definecolor{titlebg1}{HTML}{FFF8E8} % upper
%\definecolor{titlebg2}{HTML}{E4EFF8} % lower

\definecolor{titlebg1}{HTML}{F8E2DF} % aus Rot aufgehellt
\definecolor{titlebg2}{HTML}{DDF3F0} % aus Türkis aufgehellt

% Background for the first page only
\AtBeginDocument{%
  \AddToShipoutPictureBG*{%
    \begin{tikzpicture}[
      remember picture,
      overlay
    ]
      % Entire page: upper color
      \fill[titlebg1]
        (current page.south west)
        rectangle
        (current page.north east);

      % Lower area: slightly diagonal edge
      \fill[titlebg2]
        (current page.south west)
        --
        ([yshift=-0.6\paperheight]current page.north west)
        --
        ([yshift=-0.3\paperheight]current page.north east)
        --
        (current page.south east)
        -- cycle;
    \end{tikzpicture}%
  }%
}
